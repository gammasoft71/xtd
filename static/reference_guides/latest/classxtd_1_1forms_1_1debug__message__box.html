The following code example demonstrates the use of debug_message_box dialog.
#define DEBUG 1 // Force debug mode even if example is builded in release.
#include <xtd/xtd>
class form_main : public form {
public:
form_main() {
button_.parent(*this);
button_.location({10, 10});
button_.text("Error...");
button_.click += delegate_ {
debug_message_box::show(*this, "An unknown process error occured.");
};
}
static auto main() {
auto df = debug_form {};
application::run(form_main {});
}
private:
button button_;
};

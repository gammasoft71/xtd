if (control::mouse_buttons() == mouse_buttons::left) {
timer.enabled(!timer.enabled());
if (timer.enabled()) chrono.start();
else chrono.stop();

if (timer.enabled()) chrono.restart();
else chrono.reset();
label.text(" 0.0");
}
};
